%%bash
set -euo pipefail
trap 'echo "HARD STOP: grounding training cell failed on line $LINENO" >&2' ERR

commit=9aef2fe6fa6f7e3742c685eef630bf68a59f7e0b
pack_sha=dd504237bba0dd2390dbe16a631f9f42fccf4cb2cca9e496e962a66560940471
manifest_sha=16e70768e5aed2c6817613b60caa6dcac3e740b2980ee59baf94768eb0f5e249
output=/kaggle/working/grounding-train
work=$(mktemp -d /tmp/satquery-grounding.XXXXXX)
export TMPDIR="$work/tmp" HF_HOME="$work/hf" XDG_CACHE_HOME="$work/cache"
export CUDA_VISIBLE_DEVICES=0
export PIP_CACHE_DIR="$work/pip" PYTHONDONTWRITEBYTECODE=1 HF_HUB_DISABLE_IMPLICIT_TOKEN=1
mkdir -p "$TMPDIR"

[[ ! -e "$output" ]] || { echo "HARD STOP: output already exists" >&2; exit 1; }
git clone --no-checkout https://github.com/sohamsssssssssssssssss/sih2026.git "$work/repo"
git -C "$work/repo" checkout --detach "$commit"
[[ "$(git -C "$work/repo" rev-parse HEAD)" == "$commit" ]]
cd "$work/repo"
qwen_dir=$(python - <<'PYTHON'
from pathlib import Path
from kaggle.grounding_qwen_inputs import find_qwen_directory

print(find_qwen_directory(Path("/kaggle/input")))
PYTHON
)
artifact_dir() {
  python - "$1" <<'PYTHON'
from pathlib import Path
import re
import sys
from kaggle.run_grounding_baselines import digest, require

pin = sys.argv[1]
require(re.fullmatch(r"[0-9a-f]{64}", pin), "expected an ARTIFACT.json SHA-256")
matches = [file.parent for file in Path("/kaggle/input").rglob("ARTIFACT.json")
           if file.is_file() and digest(file) == pin]
require(len(matches) == 1, f"attach exactly one directory with the pinned artifact receipt (found {len(matches)})")
print(matches[0])
PYTHON
}
resume=()
if [[ -n "${SATQUERY_RESUME_RECEIPT_SHA256:-}" ]]; then
  checkpoint=$(artifact_dir "$SATQUERY_RESUME_RECEIPT_SHA256")
  resume=(--resume "$checkpoint" --resume-receipt-sha256 "$SATQUERY_RESUME_RECEIPT_SHA256")
fi
python -m venv --system-site-packages --without-pip "$work/venv"
python -m pip --python "$work/venv/bin/python" install --no-cache-dir \
  'transformers==4.49.0' 'peft==0.15.2' 'accelerate==1.5.2' 'huggingface-hub==0.30.2' 'bitsandbytes==0.48.2'
py="$work/venv/bin/python"
"$py" - <<'PYTHON'
import torch
from kaggle.run_grounding_baselines import require

require(torch.cuda.is_available() and torch.cuda.device_count() == 1 and "T4" in torch.cuda.get_device_name(0),
        "select exactly one visible T4")
PYTHON
export HF_HUB_OFFLINE=1 TRANSFORMERS_OFFLINE=1
remaining=$((39000 - SECONDS))
(( remaining > 0 )) || { echo "HARD STOP: setup consumed the session budget" >&2; exit 1; }
timeout "$remaining" "$py" -m kaggle.train_grounding_adapter --input-root /kaggle/input \
  --pack-sha256 "$pack_sha" --manifest-sha256 "$manifest_sha" --qwen-dir "$qwen_dir" \
  --output "$output" "${resume[@]}"
[[ "$(du -sb "$output" | cut -f1)" -lt 20000000000 ]]
echo "CELL DONE"
